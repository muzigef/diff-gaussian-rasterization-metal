cmake_minimum_required(VERSION 3.24)
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
endif()
project(DGR VERSION 0.1.0 LANGUAGES CXX)

option(DGR_BUILD_METAL "Build the Apple Silicon Metal backend" ${APPLE})
option(DGR_BUILD_EXAMPLES "Build the JSON/PPM command-line example" ON)
option(DGR_BUILD_TORCH "Build the compatible PyTorch MPS extension" OFF)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

add_library(dgr_core STATIC src/core/scene.cpp src/core/reference.cpp)
add_library(dgr::core ALIAS dgr_core)
set_target_properties(dgr_core PROPERTIES EXPORT_NAME core POSITION_INDEPENDENT_CODE ON)
target_compile_features(dgr_core PUBLIC cxx_std_17)
target_include_directories(dgr_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(dgr_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
set(dgr_install_targets dgr_core)

if(DGR_BUILD_METAL)
    if(NOT APPLE)
        message(FATAL_ERROR "DGR_BUILD_METAL requires Apple platforms; use OFF for the CPU reference")
    endif()
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    set(CMAKE_OBJCXX_EXTENSIONS OFF)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/rasterizer.metal" DGR_METAL_SOURCE)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/training.metal" DGR_TRAINING_SOURCE)
    string(APPEND DGR_METAL_SOURCE "\n${DGR_TRAINING_SOURCE}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/rasterizer.metal shaders/training.metal)
    configure_file(cmake/shader_source.h.in generated/shader_source.h @ONLY)
    add_library(dgr_metal STATIC src/metal/metal_rasterizer.mm)
    add_library(dgr::metal ALIAS dgr_metal)
    set_target_properties(dgr_metal PROPERTIES EXPORT_NAME metal POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(dgr_metal PUBLIC dgr_core PRIVATE "-framework Metal" "-framework Foundation")
    target_include_directories(dgr_metal PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_options(dgr_metal PRIVATE -fobjc-arc -fobjc-arc-exceptions -Wall -Wextra -Wpedantic)
    list(APPEND dgr_install_targets dgr_metal)

    if(DGR_BUILD_TORCH)
        find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.utils.cmake_prefix_path)"
            OUTPUT_VARIABLE DGR_TORCH_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        list(APPEND CMAKE_PREFIX_PATH "${DGR_TORCH_PREFIX}")
        find_package(Torch REQUIRED)
        find_library(DGR_TORCH_PYTHON torch_python PATHS "${TORCH_INSTALL_PREFIX}/lib" REQUIRED NO_DEFAULT_PATH)
        Python3_add_library(_C MODULE WITH_SOABI bindings/torch/module.cpp bindings/torch/bridge.mm)
        target_compile_features(_C PRIVATE cxx_std_20)
        set_target_properties(_C PROPERTIES OBJCXX_STANDARD 20)
        target_include_directories(_C PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" ${TORCH_INCLUDE_DIRS})
        target_link_libraries(_C PRIVATE ${TORCH_LIBRARIES} "${DGR_TORCH_PYTHON}" "-framework Metal" "-framework Foundation")
        target_compile_options(_C PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc;-fobjc-arc-exceptions>")
        if(NOT DGR_PYTHON_OUTPUT_DIR)
            set(DGR_PYTHON_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/python/diff_gaussian_rasterization")
        endif()
        set_target_properties(_C PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${DGR_PYTHON_OUTPUT_DIR}")
    endif()

    if(DGR_BUILD_EXAMPLES OR BUILD_TESTING)
        # JSON belongs to the example boundary, never to the renderer's public data/API layer.
        add_library(dgr_example_io STATIC examples/scene_io.mm)
        target_link_libraries(dgr_example_io PUBLIC dgr_core PRIVATE "-framework Foundation")
        target_include_directories(dgr_example_io PUBLIC examples)
        target_compile_options(dgr_example_io PRIVATE -fobjc-arc -fobjc-arc-exceptions -Wall -Wextra -Wpedantic)
    endif()
    if(DGR_BUILD_EXAMPLES)
        add_executable(rasterizer-demo examples/render.cpp)
        target_link_libraries(rasterizer-demo PRIVATE dgr_metal dgr_example_io)
    endif()
endif()
if(DGR_BUILD_TORCH AND NOT DGR_BUILD_METAL)
    message(FATAL_ERROR "DGR_BUILD_TORCH requires DGR_BUILD_METAL")
endif()

if(BUILD_TESTING)
    add_executable(dgr_core_tests tests/core_tests.cpp)
    target_link_libraries(dgr_core_tests PRIVATE dgr_core)
    foreach(name IN ITEMS single depth_order empty validation reference_limits)
        add_test(NAME core.${name} COMMAND dgr_core_tests ${name})
    endforeach()
    if(DGR_BUILD_METAL)
        add_executable(dgr_metal_tests tests/metal_tests.cpp)
        target_link_libraries(dgr_metal_tests PRIVATE dgr_metal)
        foreach(name IN ITEMS single_overlap empty_culled alpha tile_boundaries equal_depth scan_sizes
                              camera seeded limits lifetime moves)
            add_test(NAME metal.${name} COMMAND dgr_metal_tests ${name})
            set_tests_properties(metal.${name} PROPERTIES SKIP_RETURN_CODE 77)
        endforeach()
        add_executable(dgr_io_tests tests/io_tests.cpp)
        target_link_libraries(dgr_io_tests PRIVATE dgr_example_io)
        add_test(NAME io.schema COMMAND dgr_io_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/overlap.json")
    endif()
endif()

install(TARGETS ${dgr_install_targets} EXPORT DGRTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/dgr DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/dgr)
install(EXPORT DGRTargets NAMESPACE dgr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DGR)
configure_package_config_file(cmake/DGRConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/DGRConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DGR)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/DGRConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/DGRConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/DGRConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DGR)
